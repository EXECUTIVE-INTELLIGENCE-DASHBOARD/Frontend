type DashboardParams = Record<string, string | number>;

const REFRESH_INTERVAL = 30_000;

function formatPercentage(value: number | null | undefined): string {
  return typeof value === "number" && Number.isFinite(value)
    ? `${value.toFixed(1)}%`
    : "—";
}

function formatNumber(value: number | null | undefined): string {
  return typeof value === "number" && Number.isFinite(value)
    ? value.toLocaleString("en-US", {
        maximumFractionDigits: 1,
      })
    : "—";
}

function buildDashboardParams(filters: AcademicFilterValues): DashboardParams {
  const params: DashboardParams = {};

  const school = filters.school.trim();
  const campus = filters.campus.trim();
  const section = filters.section.trim().toLowerCase();
  const term = filters.term.trim();

  // School must be a valid database ID.
  if (school && /^\d+$/.test(school) && Number(school) > 0) {
    params.school = Number(school);
  }

  // Campus must use the backend's actual campus choice value.
  if (campus) {
    params.campus = campus;
  }

  // The backend dashboard accepts the section through "level".
  // Keep the frontend's existing section values.
  if (section) {
    params.level = section;
  }

  // The backend expects a database term ID, not the academic year.
  if (term && /^\d+$/.test(term) && Number(term) > 0) {
    params.term = Number(term);
  }

  return params;
}

function getErrorMessage(error: unknown): string {
  if (typeof error === "object" && error !== null && "response" in error) {
    const response = (
      error as {
        response?: { status?: number; data?: unknown };
      }
    ).response;

    if (response?.status === 404) {
      return "No academic data or term was found for these filters.";
    }

    if (response?.status === 400) {
      return "The selected filters were not accepted by the backend. Check the campus, section, and term values.";
    }

    if (response?.status) {
      return `Unable to load academic performance (HTTP ${response.status}).`;
    }
  }

  return "Unable to load academic performance. Check your connection and try again.";
}

export function AcademicSummary({ filters }: AcademicSummaryProps) {
  const { school, campus, section, term } = filters;

  const params = useMemo(
    () => buildDashboardParams(filters),
    [school, campus, section, term],
  );

  const dashboardQuery = useQuery({
    queryKey: ["academic-performance-dashboard", params],
    queryFn: async ({ signal }): Promise<PerformanceDashboard> => {
      const response = await api.get<PerformanceDashboard>(
        "/academics/performance/dashboard/",
        {
          params,
          signal,
        },
      );

      return response.data;
    },
    refetchInterval: REFRESH_INTERVAL,
    refetchOnWindowFocus: true,
    staleTime: 10_000,
    retry: false,
  });

  const summary = dashboardQuery.data?.summary;
  const assessmentsQuery = useQuery({
    queryKey: ["academic-assessments", params],
    queryFn: async ({ signal }): Promise<Assessment[]> => {
      const response = await api.get<Assessment[]>("/academics/assessments/", {
        params,
        signal,
      });

      return response.data;
    },
    refetchInterval: REFRESH_INTERVAL,
    refetchOnWindowFocus: true,
    staleTime: 10_000,
    retry: false,
  });
  const metrics = [
    {
      id: "assessments",
      title: "Assessments Recorded",
      icon: ClipboardList,
      value: assessmentsQuery.isPending
        ? "Loading..."
        : assessmentsQuery.isError
          ? "—"
          : formatNumber(assessmentsQuery.data?.length ?? 0),
    },
    {
      id: "performance",
      title: "Student Performance",
      icon: UsersRound,
      value: formatPercentage(summary?.student_performance),
      detail: summary
        ? `${formatNumber(summary.students_passing)} of ${formatNumber(summary.students_assessed)} students passing`
        : "Students meeting the passing mark",
    },
    {
      id: "average-grade",
      title: "Average Grade",
      icon: Award,
      value:
        summary?.average_grade != null
          ? summary.average_grade
          : formatNumber(summary?.average_mark),
      detail:
        summary?.average_grade != null
          ? `Average mark: ${formatNumber(summary.average_mark)}%`
          : "Average assessment mark (%)",
    },
  ];

  if (dashboardQuery.isPending) {
    return (
      <section
        className="acad-summary"
        aria-label="Academic performance summary"
        aria-busy="true"
      >
        {metrics.map(({ id, title }) => (
          <article className="acad-metric" key={id}>
            <div className="acad-metric__content">
              <span className="acad-metric__label">{title}</span>
              <h2 className="acad-metric__value">Loading...</h2>
              <span className="acad-metric__detail">
                Fetching live academic data
              </span>
            </div>
          </article>
        ))}
      </section>
    );
  }

  if (dashboardQuery.isError || !summary) {
    return (
      <section
        className="acad-summary"
        aria-label="Academic performance summary"
        aria-busy={dashboardQuery.isFetching}
      >
        <p role="alert">
          {dashboardQuery.isError
            ? getErrorMessage(dashboardQuery.error)
            : "The dashboard returned no summary data for these filters."}
        </p>

        <button
          type="button"
          onClick={() => void dashboardQuery.refetch()}
          disabled={dashboardQuery.isFetching}
        >
          {dashboardQuery.isFetching ? "Retrying..." : "Retry"}
        </button>
      </section>
    );
  }

  return (
    <section
      className="acad-summary"
      aria-label="Academic performance summary"
      aria-busy={dashboardQuery.isFetching}
    >
      {metrics.map(({ id, title, icon: Icon, value, detail }) => (
        <article className="acad-metric" key={id}>
          <span className="acad-metric__icon" aria-hidden="true">
            <Icon size={18} strokeWidth={1.7} />
          </span>

          <div className="acad-metric__content">
            <span className="acad-metric__label">{title}</span>
            <h2 className="acad-metric__value">{value}</h2>
            <span className="acad-metric__detail">{detail}</span>
          </div>
        </article>
      ))}
    </section>
  );
}

const REFRESH_INTERVAL = 30_000;

function getResults<T>(data: ApiResponse<T>): T[] {
  return Array.isArray(data) ? data : (data.results ?? []);
}

function formatNumber(value: number | null | undefined): string {
  return value == null ? "—" : value.toLocaleString();
}

function formatPercentage(value: number): string {
  return `${value.toFixed(1)}%`;
}

function getExamLabel(exam: NationalExamResult["exam"]): string {
  switch (exam) {
    case "ple":
      return "PLE";
    case "uce":
      return "UCE";
    case "uace":
      return "UACE";
  }
}

function getCampusLabel(school: School): string {
  if (school.campus_display) {
    return school.campus_display;
  }

  switch (school.campus.toLowerCase()) {
    case "kibuli":
      return "Kibuli Main Campus";
    case "buwaate":
      return "Buwaate Campus";
    case "annex":
      return "Annex Preschool";
    default:
      return school.campus;
  }
}

function getDivisionLabel(result: NationalExamResult): string {
  return result.division_display || result.division || "Ungraded";
}

function average(values: (number | null | undefined)[]): number | null {
  const validValues = values.filter(
    (value): value is number => value != null && Number.isFinite(value),
  );

  if (validValues.length === 0) {
    return null;
  }

  return (
    validValues.reduce((total, value) => total + value, 0) / validValues.length
  );
}

function buildExamSummaries(
  results: NationalExamResult[],
  schools: School[],
  selectedSchool: string | undefined,
): ExamSummary[] {
  const schoolsById = new Map(schools.map((school) => [school.id, school]));

  const groups = new Map<
    string,
    {
      campus: string;
      schoolName: string;
      exam: NationalExamResult["exam"];
      year: number;
      results: NationalExamResult[];
    }
  >();

  for (const result of results) {
    const school = schoolsById.get(result.school);

    if (!school) {
      continue;
    }

    const campus = getCampusLabel(school);

    // When a school is selected, report that school.
    // Otherwise, combine results across schools in the same campus.
    const schoolName = selectedSchool ? school.name : campus;

    const key = [
      selectedSchool ? school.id : school.campus,
      result.exam,
      result.year,
    ].join("-");

    const existing = groups.get(key);

    if (existing) {
      existing.results.push(result);
    } else {
      groups.set(key, {
        campus,
        schoolName,
        exam: result.exam,
        year: result.year,
        results: [result],
      });
    }
  }

  return Array.from(groups.entries())
    .map(([key, group]) => {
      const candidates = group.results.length;
      const passed = group.results.filter((result) => result.passed).length;
      const failed = candidates - passed;

      const divisionCounts = new Map<string, number>();

      for (const result of group.results) {
        const label =
          group.exam === "uace"
            ? `${result.principal_passes ?? 0} principal passes`
            : getDivisionLabel(result);

        divisionCounts.set(label, (divisionCounts.get(label) ?? 0) + 1);
      }

      return {
        key,
        campus: group.campus,
        schoolName: group.schoolName,
        exam: group.exam,
        year: group.year,
        candidates,
        passed,
        failed,
        passRate: candidates > 0 ? (passed / candidates) * 100 : 0,
        averageAggregate:
          group.exam === "ple"
            ? average(group.results.map((result) => result.aggregate))
            : null,
        averagePrincipalPasses:
          group.exam === "uace"
            ? average(group.results.map((result) => result.principal_passes))
            : null,
        averagePoints:
          group.exam === "uace"
            ? average(group.results.map((result) => result.points))
            : null,
        divisions: Array.from(divisionCounts.entries())
          .map(([label, count]) => ({ label, count }))
          .sort((a, b) =>
            a.label.localeCompare(b.label, undefined, {
              numeric: true,
            }),
          ),
      };
    })
    .sort(
      (a, b) =>
        a.campus.localeCompare(b.campus) ||
        a.exam.localeCompare(b.exam) ||
        b.year - a.year,
    );
}

export function AcademicAnalytics({ filters }: AcademicAnalyticsProps) {
  const { school, campus, section, term, year } = filters;

  const dashboardParams: Record<string, string | number> = {};

  if (school) dashboardParams.school = school;
  if (campus) dashboardParams.campus = campus;
  if (section) dashboardParams.level = section;

  if (term && /^\d+$/.test(term)) {
    dashboardParams.term = Number(term);
  }

  const dashboardQuery = useQuery({
    queryKey: ["academic-performance-dashboard", school, campus, section, term],
    queryFn: async ({ signal }) => {
      const response = await api.get<PerformanceDashboard>(
        "/academics/performance/dashboard/",
        { params: dashboardParams, signal },
      );

      return response.data;
    },
    refetchInterval: REFRESH_INTERVAL,
    refetchOnWindowFocus: true,
  });

  const schoolsQuery = useQuery({
    queryKey: ["academic-national-exam-schools"],
    queryFn: async ({ signal }) => {
      const response = await api.get<ApiResponse<School>>(
        "/academics/schools/",
        { signal },
      );

      return getResults(response.data);
    },
    refetchInterval: REFRESH_INTERVAL,
    refetchOnWindowFocus: true,
  });

  const examsQuery = useQuery({
    queryKey: ["academic-national-exam-results", school, campus, section, year],
    queryFn: async ({ signal }) => {
      const params: Record<string, string> = {};

      if (school) params.school = school;
      if (campus) params.campus = campus;
      if (section) params.level = section;
      if (year) params.year = year;

      const response = await api.get<ApiResponse<NationalExamResult>>(
        "/academics/national-exam-results/",
        { params, signal },
      );

      return getResults(response.data);
    },
    refetchInterval: REFRESH_INTERVAL,
    refetchOnWindowFocus: true,
  });

  const schools = schoolsQuery.data ?? [];
  const examResults = examsQuery.data ?? [];

  const examSummaries = useMemo(
    () => buildExamSummaries(examResults, schools, school),
    [examResults, schools, school],
  );

  if (dashboardQuery.isLoading) {
    return (
      <section className="acad-analytics" aria-busy="true">
        <p>Loading academic analytics...</p>
      </section>
    );
  }

  if (dashboardQuery.isError || !dashboardQuery.data) {
    return (
      <section className="acad-analytics" aria-busy="false">
        <p role="alert">Unable to load academic analytics. Please try again.</p>
      </section>
    );
  }

  const { grade_distribution } = dashboardQuery.data;

  return (
    <section className="acad-analytics" aria-label="Academic analytics">
      <article className="acad-analytics__card">
        <header className="acad-analytics__header">
          <div>
            <h2>Grade Distribution</h2>
          </div>
        </header>

        {grade_distribution.length === 0 ? (
          <p>No grade distribution data is available for these filters.</p>
        ) : (
          <div className="acad-analytics__grade-columns">
            {grade_distribution.map((level) => (
              <div className="acad-analytics__grade-level" key={level.level}>
                <div className="acad-analytics__grade-heading">
                  <h3>{level.level_display}</h3>
                  <span>{level.students} students</span>
                </div>

                <div className="acad-analytics__grade-list">
                  {level.grades.map((grade) => {
                    const percentage =
                      level.students > 0
                        ? (grade.students / level.students) * 100
                        : 0;

                    return (
                      <div
                        className="acad-analytics__grade-row"
                        key={grade.grade}
                      >
                        <span className="acad-analytics__grade-label">
                          {grade.grade}
                        </span>

                        <div
                          className="acad-analytics__grade-track"
                          role="progressbar"
                          aria-label={`${level.level_display} grade ${grade.grade}`}
                          aria-valuemin={0}
                          aria-valuemax={level.students}
                          aria-valuenow={grade.students}
                        >
                          <span
                            className="acad-analytics__grade-fill"
                            style={{ width: `${percentage}%` }}
                          />
                        </div>

                        <span className="acad-analytics__grade-count">
                          {grade.students}
                        </span>
                      </div>
                    );
                  })}
                </div>
              </div>
            ))}
          </div>
        )}
      </article>

      {/* National Examination Results — dynamically grouped backend data */}
      <article className="acad-analytics__card">
        <header className="acad-analytics__header">
          <div>
            <h2>National Examination Results</h2>
          </div>
        </header>

        {schoolsQuery.isLoading || examsQuery.isLoading ? (
          <p>Loading national examination results...</p>
        ) : schoolsQuery.isError || examsQuery.isError ? (
          <p role="alert">
            Unable to load national examination results. Please try again.
          </p>
        ) : examSummaries.length === 0 ? (
          <p>
            No national examination results are available for the selected
            filters.
          </p>
        ) : (
          <div className="acad-exams">
            <div className="acad-exams__table-wrap">
              <table className="acad-exams__table">
                <thead>
                  <tr>
                    <th scope="col">Campus</th>
                    <th scope="col">Exam</th>
                    <th scope="col">Year</th>
                    <th scope="col">Candidates</th>
                    <th scope="col">Passed</th>
                    <th scope="col">Failed</th>
                    <th scope="col">Pass Rate</th>
                  </tr>
                </thead>

                <tbody>
                  {examSummaries.map((summary) => (
                    <tr key={summary.key}>
                      <td>{summary.schoolName}</td>
                      <td>{getExamLabel(summary.exam)}</td>
                      <td>{summary.year}</td>
                      <td>{formatNumber(summary.candidates)}</td>
                      <td>{formatNumber(summary.passed)}</td>
                      <td>{formatNumber(summary.failed)}</td>
                      <td>
                        <span className="acad-exams__pass-rate">
                          {formatPercentage(summary.passRate)}
                        </span>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}
      </article>
    </section>
  );
}

function isObject(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null;
}

function toArray<T>(value: unknown): T[] {
  if (Array.isArray(value)) {
    return value as T[];
  }

  if (isObject(value) && Array.isArray(value.results)) {
    return value.results as T[];
  }

  return [];
}

function toNumber(value: unknown): number | null {
  if (typeof value === "number" && Number.isFinite(value)) {
    return value;
  }

  if (typeof value === "string" && value.trim() !== "") {
    const parsed = Number(value);
    return Number.isFinite(parsed) ? parsed : null;
  }

  return null;
}

function formatNumber(value: number): string {
  return value.toLocaleString("en-UG");
}

function formatPercent(value: number | null): string {
  return value === null ? "—" : `${value.toFixed(1)}%`;
}

function errorMessage(error: unknown): string {
  if (
    isObject(error) &&
    isObject(error.response) &&
    typeof error.response.status === "number"
  ) {
    return `Unable to load data (HTTP ${error.response.status}).`;
  }

  return "Unable to load data. Please try again.";
}

function normalizeLevel(value: string | undefined): string {
  const level = (value ?? "").trim().toLowerCase().replace(/[- ]/g, "_");

  if (["primary", "primary_level"].includes(level)) return "Primary";
  if (["o_level", "olevel", "ordinary_level"].includes(level)) return "O Level";
  if (["a_level", "alevel", "advanced_level"].includes(level)) return "A Level";
  if (level.includes("nursery")) return "Nursery";

  return value?.trim() || "Other";
}

function normalizeCategory(value: string | undefined): string {
  const category = (value ?? "").trim().toLowerCase();

  if (category === "science" || category === "sciences") return "Sciences";
  if (category === "arts" || category === "art") return "Arts";

  return "";
}

function getSubjectRows(subjects: DashboardSubject[]): SubjectRow[] {
  const grouped = new Map<
    string,
    {
      key: string;
      subject: string;
      level: string;
      markTotal: number;
      markCount: number;
      passTotal: number;
      passCount: number;
      previousTotal: number;
      previousCount: number;
    }
  >();

  for (const item of subjects) {
    const level = normalizeLevel(item.level_display || item.level);
    const category = normalizeCategory(item.category);

    // Primary subjects remain individual. Secondary subjects are grouped
    // into Sciences and Arts, using the backend's category field.
    let subjectName = item.subject || item.name || item.code || "Subject";

    if (level === "O Level" || level === "A Level") {
      subjectName = category || "Uncategorised";
    }

    const key = `${level}:${subjectName}`;

    if (!grouped.has(key)) {
      grouped.set(key, {
        key,
        subject: subjectName,
        level,
        markTotal: 0,
        markCount: 0,
        passTotal: 0,
        passCount: 0,
        previousTotal: 0,
        previousCount: 0,
      });
    }

    const row = grouped.get(key)!;
    const mark = toNumber(item.average_mark);
    const pass = toNumber(item.pass_rate);
    const previous = toNumber(item.previous_pass_rate);

    if (mark !== null) {
      row.markTotal += mark;
      row.markCount += 1;
    }

    if (pass !== null) {
      row.passTotal += pass;
      row.passCount += 1;
    }

    if (previous !== null) {
      row.previousTotal += previous;
      row.previousCount += 1;
    }
  }

  return Array.from(grouped.values())
    .map((row) => ({
      key: row.key,
      subject: row.subject,
      level: row.level,
      averageMark: row.markCount > 0 ? row.markTotal / row.markCount : null,
      passRate: row.passCount > 0 ? row.passTotal / row.passCount : null,
      previousPassRate:
        row.previousCount > 0 ? row.previousTotal / row.previousCount : null,
    }))
    .sort((a, b) => {
      const order = ["Nursery", "Primary", "O Level", "A Level", "Other"];
      const difference = order.indexOf(a.level) - order.indexOf(b.level);

      return difference || a.subject.localeCompare(b.subject);
    });
}

function getCompletionRecords(
  data: DashboardResponse | undefined,
): CompletionRecord[] {
  if (!data) return [];

  // Only use completion data actually returned by the backend.
  for (const key of ["completion", "student_completion", "completion_rates"]) {
    const records = toArray<CompletionRecord>(data[key]);
    if (records.length > 0) return records;
  }

  return [];
}

function getCompletedCount(record: CompletionRecord): number | null {
  return toNumber(record.completed_students ?? record.completed);
}

function getCompletionRate(record: CompletionRecord): number | null {
  return toNumber(record.completion_rate ?? record.completion_percentage);
}

function getQueryParams(filters: AcademicFilterValues) {
  const params: Record<string, string> = {};

  if (filters.school && filters.school !== "All Schools") {
    params.school = filters.school;
  }

  if (filters.campus && filters.campus !== "All Campuses") {
    params.campus = filters.campus;
  }

  if (filters.term) params.term = filters.term;
  if (filters.year) params.year = filters.year;

  const section = filters.section.trim().toLowerCase();

  if (section === "primary") {
    params.level = "primary";
  }

  return params;
}

export function AcademicDetails({
  filters = DEFAULT_FILTERS,
}: AcademicDetailsProps) {
  const queryParams = useMemo(() => getQueryParams(filters), [filters]);

  const dashboardQuery = useQuery({
    queryKey: ["academic-details-dashboard", queryParams],
    queryFn: async ({ signal }) => {
      const response = await api.get<DashboardResponse>(
        "/academics/performance/dashboard/",
        {
          params: queryParams,
          signal,
        },
      );

      return response.data;
    },
    refetchInterval: REFRESH_INTERVAL,
    refetchOnWindowFocus: true,
  });

  const subjects = useMemo(
    () => toArray<DashboardSubject>(dashboardQuery.data?.subjects),
    [dashboardQuery.data],
  );

  const subjectRows = useMemo(() => getSubjectRows(subjects), [subjects]);

  const completionRecords = useMemo(
    () => getCompletionRecords(dashboardQuery.data),
    [dashboardQuery.data],
  );

  const selectedSection = filters.section.trim().toLowerCase();

  const visibleSubjectRows = subjectRows.filter((row) => {
    if (selectedSection === "primary") return row.level === "Primary";
    if (selectedSection === "nursery") return row.level === "Nursery";
    if (selectedSection === "secondary") {
      return row.level === "O Level" || row.level === "A Level";
    }

    return true;
  });

  return (
    <section
      className="acad-details"
      aria-label="Subject comparison and student completion rate"
      aria-busy={dashboardQuery.isFetching}
    >
      <div className="acad-details__grid">
        <article className="acad-details__card">
          <header className="acad-details__header">
            <div>
              <h2>Subject Comparison</h2>
            </div>
            {dashboardQuery.isFetching && !dashboardQuery.isLoading && (
              <span className="acad-details__live">
                <span className="acad-details__live-dot" />
                Updating
              </span>
            )}
          </header>

          {dashboardQuery.isLoading ? (
            <p className="acad-details__message" role="status">
              Loading subject performance…
            </p>
          ) : dashboardQuery.isError ? (
            <p
              className="acad-details__message acad-details__message--error"
              role="alert"
            >
              {errorMessage(dashboardQuery.error)}
            </p>
          ) : visibleSubjectRows.length === 0 ? (
            <p className="acad-details__message">
              No subject performance data is available for the selected filters.
            </p>
          ) : (
            <div className="acad-details__table-scroll">
              <table className="acad-details__table">
                <thead>
                  <tr>
                    <th scope="col">Subject / Category</th>
                    <th scope="col">Level</th>
                    <th scope="col">Average mark</th>
                    <th scope="col">Pass rate</th>
                  </tr>
                </thead>
                <tbody>
                  {visibleSubjectRows.map((row) => {
                    return (
                      <tr key={row.key}>
                        <td className="acad-details__subject">{row.subject}</td>
                        <td>{row.level}</td>
                        <td>{formatPercent(row.averageMark)}</td>
                        <td className="acad-details__strong">
                          {formatPercent(row.passRate)}
                        </td>
                      </tr>
                    );
                  })}
                </tbody>
              </table>
            </div>
          )}
        </article>

        <article className="acad-details__card">
          <header className="acad-details__header">
            <div>
              <h2>Student Completion Rate</h2>
            </div>
          </header>

          {dashboardQuery.isLoading ? (
            <p className="acad-details__message" role="status">
              Loading completion data…
            </p>
          ) : dashboardQuery.isError ? (
            <p
              className="acad-details__message acad-details__message--error"
              role="alert"
            >
              {errorMessage(dashboardQuery.error)}
            </p>
          ) : completionRecords.length === 0 ? (
            <div className="acad-details__empty-state">
              <p>
                No student-completion records were returned by the dashboard
                API.
              </p>
            </div>
          ) : (
            <div className="acad-details__completion-list">
              {completionRecords.map((record, index) => {
                const level =
                  record.level_display || record.level || "Education level";
                const campus = record.campus_display || record.campus || "";
                const completed = getCompletedCount(record);
                const rate = getCompletionRate(record);

                return (
                  <div
                    className="acad-details__completion-row"
                    key={`${level}-${campus}-${index}`}
                  >
                    <div className="acad-details__completion-top">
                      <div>
                        <strong>{normalizeLevel(level)}</strong>
                        {campus && (
                          <span className="acad-details__campus">{campus}</span>
                        )}
                      </div>
                      <strong className="acad-details__completion-rate">
                        {formatPercent(rate)}
                      </strong>
                    </div>

                    <div className="acad-details__progress-track">
                      <span
                        style={{
                          width: `${Math.min(100, Math.max(0, rate ?? 0))}%`,
                        }}
                      />
                    </div>

                    <p className="acad-details__completed">
                      {completed === null
                        ? "Completed students: not provided by the API"
                        : `${formatNumber(completed)} completed students`}
                    </p>
                  </div>
                );
              })}
            </div>
          )}
        </article>
      </div>
    </section>
  );
}
